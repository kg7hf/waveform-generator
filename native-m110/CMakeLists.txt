include("${CMAKE_CURRENT_SOURCE_DIR}/sources.cmake")

set(WFG_NATIVE_HOST_SOURCES ${WFG_NATIVE_M110_SOURCES})
list(FILTER WFG_NATIVE_HOST_SOURCES EXCLUDE REGEX "/waveform-source/")
add_library(wfg_native_m110 STATIC ${WFG_NATIVE_HOST_SOURCES})
target_include_directories(wfg_native_m110 PUBLIC "${WFG_ROOT}" "${WFG_NATIVE_M110_INCLUDE}")
target_link_libraries(wfg_native_m110 PUBLIC wfg_waveform_source)
target_compile_features(wfg_native_m110 PUBLIC cxx_std_23)
if(MSVC)
    target_compile_options(wfg_native_m110 PRIVATE /W4 /WX /permissive- /fp:strict)
else()
    target_compile_options(wfg_native_m110 PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
endif()

add_executable(wfg_native_m110_tests tests.cpp)
target_link_libraries(wfg_native_m110_tests PRIVATE wfg_native_m110)
target_compile_features(wfg_native_m110_tests PRIVATE cxx_std_23)
if(MSVC)
    target_compile_options(wfg_native_m110_tests PRIVATE /W4 /WX /permissive- /fp:strict)
else()
    target_compile_options(wfg_native_m110_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
endif()
set_target_properties(wfg_native_m110_tests PROPERTIES ENABLE_EXPORTS OFF)
add_custom_command(TARGET wfg_native_m110_tests POST_BUILD COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_CURRENT_BINARY_DIR}/libwfg_native_m110_tests.dll.a")
add_test(NAME wfg-native-m110 COMMAND wfg_native_m110_tests)
